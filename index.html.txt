// ── Board Init ────────────────────────────────────────────────────────────
async function initBoard() {
  await loadCards()
  subscribeRealtime()
}

// ── Load cards ────────────────────────────────────────────────────────────
async function loadCards() {
  const { data, error } = await db.from('cards').select('*').order('position').order('created_at')
  if (error) { showToast('Erro ao carregar: ' + error.message); return }
  cards = data || []
  renderBoard()
}

// ── Realtime sync ─────────────────────────────────────────────────────────
function subscribeRealtime() {
  db.channel('cards-changes')
    .on('postgres_changes', { event: '*', schema: 'public', table: 'cards' }, () => loadCards())
    .subscribe()
}

  board.innerHTML = ''
  document.getElementById('total-count').textContent = cards.length + ' tarefa' + (cards.length !== 1 ? 's' : '')

function buildCard(card) {
  const el = document.createElement('div')
  el.className = 'card'
  el.draggable = true
  el.dataset.id = card.id

  const progColor = card.progress >= 100 ? '#10B981' : card.progress >= 60 ? '#6366F1' : card.progress >= 30 ? '#F59E0B' : '#F87171'
  const statusCls = { 'Entrada':'inbox','Fazendo':'doing','Concluído':'done','Bloqueado':'blocked' }[card.status] || 'inbox'
  const prioCls = { 'Alta':'alta','Média':'media','Baixa':'baixa' }[card.priority] || 'media'

  const deadlineHtml = (() => {
    if (!card.deadline) return ''
    const d = new Date(card.deadline + 'T00:00:00')
    const today = new Date(); today.setHours(0,0,0,0)
    const overdue = d < today && card.status !== 'Concluído'
    const fmt = d.toLocaleDateString('pt-BR', { day:'2-digit', month:'short' })
    return `<span class="card-deadline${overdue?' overdue':''}">${overdue ? '⚠ ' : ''}${fmt}</span>`
  })()

  const initials = card.owner ? card.owner.split(' ').map(w=>w[0]).join('').slice(0,2).toUpperCase() : ''

  el.innerHTML = `
    <div class="card-menu">
      <button class="card-menu-btn" onclick="editCard('${card.id}')" title="Editar">✏</button>
      <button class="card-menu-btn" onclick="deleteCard('${card.id}')" title="Excluir" style="color:#991B1B">✕</button>
    </div>
    <div class="card-title">${escHtml(card.title)}</div>
    <div class="card-tags">
      <span class="tag tag-status-${statusCls}">${card.status}</span>
      ${card.priority ? `<span class="tag tag-prio-${prioCls}">${card.priority}</span>` : ''}
      ${card.category ? `<span class="tag" style="background:#F4F3F0;color:#706E68">${escHtml(card.category)}</span>` : ''}
    </div>
    <div class="card-footer">
      ${card.owner ? `<div class="card-owner"><div class="avatar-sm">${initials}</div>${escHtml(card.owner)}</div>` : ''}
      <div class="card-spacer"></div>
      ${deadlineHtml}
    </div>
    ${card.progress > 0 ? `
      <div class="prog-wrap">
        <div class="prog-bar"><div class="prog-fill" style="width:${card.progress}%;background:${progColor}"></div></div>
        <div class="prog-label">${card.progress}%</div>
      </div>` : ''}
  `

  el.addEventListener('dragstart', () => { dragId = card.id; setTimeout(() => el.classList.add('dragging'), 0) })
  el.addEventListener('dragend', () => { el.classList.remove('dragging'); dragId = null })
  return el
}

// ── Drag & Drop ───────────────────────────────────────────────────────────
function setupDropZone(area, status) {
  area.addEventListener('dragover', e => { e.preventDefault(); area.classList.add('drag-over') })
  area.addEventListener('dragleave', () => area.classList.remove('drag-over'))
  area.addEventListener('drop', async e => {
    e.preventDefault()
    area.classList.remove('drag-over')
    if (!dragId) return
    const card = cards.find(c => c.id === dragId)
    if (!card || card.status === status) return
    await db.from('cards').update({ status }).eq('id', dragId)
    showToast('Card movido para ' + status)
  })
}

// ── CRUD ──────────────────────────────────────────────────────────────────
function openModal(status) {
  defaultStatus = status || 'Entrada'
  editingId = null
  document.getElementById('modal-title').textContent = 'Novo card'
  document.getElementById('f-title').value = ''
  document.getElementById('f-desc').value = ''
  document.getElementById('f-status').value = defaultStatus
  document.getElementById('f-priority').value = 'Média'
  document.getElementById('f-owner').value = currentUser
  document.getElementById('f-deadline').value = ''
  document.getElementById('f-category').value = ''
  document.getElementById('f-progress').value = 0
  document.getElementById('prog-val').textContent = '0%'
  document.getElementById('modal').classList.add('open')
  setTimeout(() => document.getElementById('f-title').focus(), 50)
}

function editCard(id) {
  const c = cards.find(x => x.id === id)
  if (!c) return
  editingId = id
  document.getElementById('modal-title').textContent = 'Editar card'
  document.getElementById('f-title').value = c.title || ''
  document.getElementById('f-desc').value = c.description || ''
  document.getElementById('f-status').value = c.status || 'Entrada'
  document.getElementById('f-priority').value = c.priority || 'Média'
  document.getElementById('f-owner').value = c.owner || ''
  document.getElementById('f-deadline').value = c.deadline || ''
  document.getElementById('f-category').value = c.category || ''
  document.getElementById('f-progress').value = c.progress || 0
  document.getElementById('prog-val').textContent = (c.progress || 0) + '%'
  document.getElementById('modal').classList.add('open')
}

async function saveCard() {
  const title = document.getElementById('f-title').value.trim()
  if (!title) { document.getElementById('f-title').focus(); return }

  const payload = {
    title,
    description: document.getElementById('f-desc').value.trim(),
    status:    document.getElementById('f-status').value,
    priority:  document.getElementById('f-priority').value,
    owner:     document.getElementById('f-owner').value.trim(),
    deadline:  document.getElementById('f-deadline').value || null,
    category:  document.getElementById('f-category').value.trim(),
    progress:  parseInt(document.getElementById('f-progress').value),
  }

  closeModal()
  if (editingId) {
    const { error } = await db.from('cards').update(payload).eq('id', editingId)
    if (error) showToast('Erro ao salvar'); else showToast('Card atualizado ✓')
  } else {
    const { error } = await db.from('cards').insert(payload)
    if (error) showToast('Erro ao criar'); else showToast('Card criado ✓')
  }
}

async function deleteCard(id) {
  if (!confirm('Excluir este card?')) return
  const { error } = await db.from('cards').delete().eq('id', id)
  if (error) showToast('Erro ao excluir'); else showToast('Card excluído')
}

function closeModal() { document.getElementById('modal').classList.remove('open') }

document.getElementById('modal').addEventListener('click', e => {
  if (e.target === document.getElementById('modal')) closeModal()
})
document.addEventListener('keydown', e => { if (e.key === 'Escape') closeModal() })

// ── Toast ─────────────────────────────────────────────────────────────────
function showToast(msg) {
  const t = document.getElementById('toast')
  t.textContent = msg
  t.classList.add('show')
  setTimeout(() => t.classList.remove('show'), 2500)
}

// ── Utils ─────────────────────────────────────────────────────────────────
function escHtml(s) {
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;')
}
